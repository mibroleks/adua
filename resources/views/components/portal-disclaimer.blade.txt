    @push('scripts')
        <script>
            /*
            |--------------------------------------------------------------------------
            | Admissions Portal Disclaimer Controller
            |--------------------------------------------------------------------------
            | Phase 1:
            | - Shows disclaimer when no acknowledgement exists.
            | - Stores acknowledgement in localStorage.
            | - Keeps behavior isolated for future settings integration.
            */

            document.addEventListener('DOMContentLoaded', function () {
                const disclaimer =
                    document.querySelector('[data-disclaimer]');

                if (!disclaimer) {
                    return;
                }

                const enabled =
                    disclaimer.dataset.enabled === 'true';

                if (!enabled) {
                    return;
                }

                const storageKey =
                    disclaimer.dataset.storageKey ||
                    'admissions_disclaimer_acknowledged';

                const continueButton =
                    disclaimer.querySelector(
                        '[data-disclaimer-continue]'
                    );

                const dialog =
                    disclaimer.querySelector(
                        '[data-disclaimer-dialog]'
                    );

                const backdrop =
                    disclaimer.querySelector(
                        '[data-disclaimer-backdrop]'
                    );

                if (!continueButton || !dialog) {
                    return;
                }

                let previouslyFocusedElement = null;

                const hasAcknowledged =
                    localStorage.getItem(storageKey) === 'true';

                if (hasAcknowledged) {
                    return;
                }

                previouslyFocusedElement =
                    document.activeElement;

                document.body.classList.add(
                    'portal-disclaimer-open'
                );

                disclaimer.hidden = false;

                requestAnimationFrame(function () {
                    dialog.focus();
                });

                function closeDisclaimer() {
                    localStorage.setItem(
                        storageKey,
                        'true'
                    );

                    disclaimer.classList.add(
                        'is-closing'
                    );

                    document.body.classList.remove(
                        'portal-disclaimer-open'
                    );

                    window.setTimeout(function () {
                        disclaimer.hidden = true;
                        disclaimer.classList.remove(
                            'is-closing'
                        );

                        if (
                            previouslyFocusedElement &&
                            typeof previouslyFocusedElement.focus ===
                                'function'
                        ) {
                            previouslyFocusedElement.focus();
                        }
                    }, 180);
                }

                continueButton.addEventListener(
                    'click',
                    closeDisclaimer
                );

                /*
                |--------------------------------------------------------------------------
                | Keyboard handling
                |--------------------------------------------------------------------------
                |
                | Escape is intentionally not used to dismiss the disclaimer.
                | The applicant must explicitly acknowledge the notice.
                |
                */

                document.addEventListener(
                    'keydown',
                    function (event) {
                        if (
                            disclaimer.hidden ||
                            event.key !== 'Tab'
                        ) {
                            return;
                        }

                        const focusableElements =
                            disclaimer.querySelectorAll(
                                'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
                            );

                        if (!focusableElements.length) {
                            return;
                        }

                        const first =
                            focusableElements[0];

                        const last =
                            focusableElements[
                                focusableElements.length - 1
                            ];

                        if (
                            event.shiftKey &&
                            document.activeElement === first
                        ) {
                            event.preventDefault();
                            last.focus();
                        } else if (
                            !event.shiftKey &&
                            document.activeElement === last
                        ) {
                            event.preventDefault();
                            first.focus();
                        }
                    }
                );
            });
        </script>
